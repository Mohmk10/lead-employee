EMPLOYEE (
  id        BIGINT PRIMARY KEY,
  nom       TEXT NOT NULL,
  tel       TEXT NOT NULL UNIQUE,
  salaire   NUMERIC(12,2) NOT NULL CHECK (salaire >= 0)
);


ADMIN (
  id_employee  BIGINT PRIMARY KEY REFERENCES EMPLOYEE(id) ON DELETE RESTRICT ON UPDATE RESTRICT
);


MANAGER (
  id_employee  BIGINT PRIMARY KEY REFERENCES EMPLOYEE(id) ON DELETE RESTRICT ON UPDATE RESTRICT,
  prime        NUMERIC(12,2) NOT NULL CHECK (prime >= 0),
  service_id   BIGINT NOT NULL UNIQUE REFERENCES SERVICE(id) ON DELETE RESTRICT ON UPDATE RESTRICT
);


DEVELOPER (
  id_employee  BIGINT PRIMARY KEY REFERENCES EMPLOYEE(id) ON DELETE RESTRICT ON UPDATE RESTRICT,
  specialite   TEXT NOT NULL CHECK (specialite IN ('FULLSTACK','FRONTEND','BACKEND')),
  service_id   BIGINT NOT NULL REFERENCES SERVICE(id) ON DELETE RESTRICT ON UPDATE RESTRICT
);


SERVICE (
  id        BIGINT PRIMARY KEY,
  libelle   TEXT NOT NULL UNIQUE
);
